```typescript
export function subscribeToJob(
  jobId: string,
  callback: (job: ScanJob) => void
): RealtimeChannel {
  const channel = supabase
    .channel(`scan_job_${jobId}`)
    .on(
      'postgres_changes',
      {
        event: 'UPDATE',
        schema: 'public',
        table: 'scan_jobs',
        filter: `id=eq.${jobId}`
      },
      (payload) => {
        if (payload.eventType === 'UPDATE' && payload.new) {
          callback(payload.new as ScanJob);
        }
      }
    )
    .subscribe();
  return channel;
}
```

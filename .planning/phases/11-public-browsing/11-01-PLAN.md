    Add to `src/features/recipes/search.ts` (do NOT modify existing `searchRecipes` or other exports):

    ```typescript
    export type PublicBrowseCursor = { page: number };
    export type PublicBrowseFilters = {
      query?: string;
      tag?: string;       // single tag chip — ignored when 'All'
      cursor?: PublicBrowseCursor;
      pageSize?: number;  // default 20
    };
    export type PublicBrowsePage = {
      recipes: Recipe[];
      hasMore: boolean;
      nextCursor: PublicBrowseCursor | null;
    };

    export async function searchPublicRecipes(filters: PublicBrowseFilters = {}): Promise<PublicBrowsePage>
    ```

    Implementation details:
    - pageSize defaults to 20
    - page from cursor defaults to 0
    - `from = page * pageSize`, `to = from + pageSize` (fetch pageSize+1 to detect hasMore)
    - `.eq('visibility', 'public')` always applied
    - `.order('created_at', { ascending: false })`
    - `.ilike('title', '%query%')` when query is non-empty after trim
    - `.overlaps('tags', [tag])` when tag is truthy and NOT 'All'
    - `.range(from, from + pageSize)` — Supabase range is inclusive, fetching pageSize+1 rows
    - hasMore = items.length > pageSize
    - recipes = hasMore ? items.slice(0, pageSize) : items
    - nextCursor = hasMore ? { page: page + 1 } : null
